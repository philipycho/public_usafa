# ---- Keep the columns we need; runs-before-play and half-inning id ----
pbp <- retro2016 |>
  select(gid, inning, top_bot, batter, pitcher, score_v, score_h, pa,
         single, double, triple, hr, hbp, walk, k,
         sb2, cs2, bip, roe, fc, sf, sh,
         outs_pre, outs_post,
         br1_pre, br2_pre, br3_pre,
         br1_post, br2_post, br3_post,
         runs, pn, gametype) |>
  mutate(runs_pre = score_v + score_h,
         half_inning_id = paste(gid, inning, top_bot)) |>
  filter(gametype == "regular")

# ---- One row per half-inning: max outs recorded and the final game score ----
half_innings <- pbp |>
  group_by(half_inning_id) |>
  summarize(max_outs   = max(outs_post),
            runs_final = max(runs_pre + runs),
            .groups = "drop")

# ---- Base-out state before and after each play; drop non-transitions ----
pbp <- pbp |>
  inner_join(half_innings, by = "half_inning_id") |>
  mutate(base_out_pre  = paste(as.integer(!is.na(br1_pre)),
                               as.integer(!is.na(br2_pre)),
                               as.integer(!is.na(br3_pre)), outs_pre,  sep = ""),
         base_out_post = paste(as.integer(!is.na(br1_post)),
                               as.integer(!is.na(br2_post)),
                               as.integer(!is.na(br3_post)), outs_post, sep = "")) |>
  filter(base_out_pre != base_out_post | runs > 0)

# ---- Complete innings only, then runs to end of inning (ROI) ----
complete_pbp <- pbp |>
  filter(max_outs > 2) |>
  mutate(runs_roi = runs_final - runs_pre)

# ---- The run expectancy matrix ----
RE24 <- complete_pbp |>
  group_by(base_out_pre) |>
  summarize(expected_runs = mean(runs_roi), .groups = "drop") |>
  arrange(expected_runs)

# ---- Attach RE before/after, then the run value of every play + an outcome label ----
complete_pbp <- complete_pbp |>
  left_join(RE24 |> rename(RE24_pre  = expected_runs), by = "base_out_pre") |>
  left_join(RE24 |> rename(RE24_post = expected_runs),
            by = join_by(base_out_post == base_out_pre)) |>
  mutate(RE24_post = coalesce(RE24_post, 0),
         run_value = RE24_post - RE24_pre + runs,
         outcome = case_when(
           single == 1 ~ "single",
           double == 1 ~ "double",
           triple == 1 ~ "triple",
           hr     == 1 ~ "hr",
           hbp    == 1 ~ "hbp",
           walk   == 1 ~ "walk",
           k      == 1 ~ "k",
           sf     == 1 ~ "sac_fly",
           sh     == 1 ~ "sac_bunt",
           bip == 1 & roe == 0 & fc == 0 ~ "bip_out",
           TRUE        ~ "other"))

# ---- Linear weights: average run value per event type ----
linear_weights <- complete_pbp |>
  filter(pa == 1) |>
  group_by(outcome) |>
  summarise(mean_run_value = mean(run_value), n = n(), .groups = "drop") |>
  arrange(desc(mean_run_value))

Expected Run Value by Count (Section 6.2.3)

We reformat RE24 to match this state format, join it twice to compute the run value of each play, and assign three-out states zero expected runs. Then retrosheet_add_counts() adds 12 logical columns, c00 through c32, indicating whether the plate appearance passed through each count.

erm2016 <- RE24 |> 
  rename(state = base_out_pre) |>
  mutate(state = str_replace(state, "^(.{3})(.)", "\\1 \\2"))

temp_df <- pbp2016 |> 
  # merge RE24 based on pre and post state
  left_join(erm2016, join_by(state == state)) |>
  rename(expected_runs_pre = expected_runs) |>
  left_join(erm2016, join_by(new_state == state)) |>
  rename(expected_runs_post = expected_runs) |>
  # fill in 0s exp runs for 3-out states
  mutate(expected_runs_post = coalesce(expected_runs_post, 0)) |>
  # get run value of each play
  mutate(run_value = expected_runs_post - expected_runs_pre + runs_scored) |>
  retrosheet_add_counts()

The Importance of the Previous Count (Section 6.2.4)

So far, a 2-2 count is worth the same whether it was reached from 0-2 or 2-0. However, a batter ahead 2-0 can afford to take a strike, so one might expect that path to favor the batter. We test this, beginning with all plate appearances that passed through 2-2.

count22 <- pbp2016 |>
  filter(c22 == 1)

count22 |>
  summarize(
    num_ab = n(), 
    mean_run_value = mean(run_value)
  )

  num_ab mean_run_value

  group_by(after2) |>

Contrary to expectation, the 0-2 path (−0.031) is slightly better for the batter than the 2-0 path (−0.042). However, most 0-2 plate appearances end before reaching 2-2, so those that survive are not a random sample. Thus, the comparison reflects selection as much as the path itself.

Extension: Connecting to Chapter 5

In Chapter 5, we credited each batter with the run value of every plate appearance, yet expected runs change with every pitch. Answer each question directly below it.

Crediting the Batter

  1. A batter takes a first-pitch ball. According to the count values above, the batting team gained approximately 0.04 expected runs. Who earned those runs: the batter, the pitcher, or the umpire? What data would you need to decide?

  2. In Chapter 5, a single with the bases empty and no outs receives the same run value whether it comes on an 0-2 count or a 3-1 count. Is that fair to the batter? Is it fair to the pitcher?

  3. Suppose each pitch before the last earns the change in count value it produced, and the final pitch earns the run value minus the value of the final count. Would any batter’s season total differ from Chapter 5? Prove it.

  4. If the totals do not change, what, if anything, have we gained by crediting pitch by pitch?

Linear Weights by Count

  1. In Chapter 5, a strikeout was worth approximately −0.26 runs. Is a strikeout on an 0-2 count as damaging as a strikeout on a 3-2 count? Before you measure it, decide what “damaging” should be measured against.
  1. Is a walk on a 3-0 count worth the same as a walk on a 3-2 count? Should a batter receive any credit for an intentional walk?

A Finer State Space

  1. The RE24 matrix conditions on 24 base-out states, while the count values above average over all of them. Is a 3-0 count worth the same with the bases empty and two outs as with the bases loaded and no outs? Estimate both.
  1. Combining 24 base-out states with 12 counts gives 288 states. How many 2016 plate appearances passed through the rarest one, and what does that imply for estimating a 288-state matrix? What would you do about it?

Batter or Pitcher?

  1. In Section 6.2.4, the plate appearances that reached 2-2 from 0-2 were not a random sample. Which types of batters are over-represented? Which types of pitchers?

  2. RE24 run value is zero-sum: every run the batter gains, the pitcher loses. Is count value zero-sum between the batter and the pitcher? Is anyone else involved?

  3. Over a season, each batter faces many different pitchers. Under what assumption does that make count-based credit a fair measure of the batter? Describe a specific situation in which that assumption fails.

  4. Can we use count values to evaluate batters at all, or is the count too entangled with the pitcher? Take a position and defend it.
